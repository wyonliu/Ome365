## 一 · 员工维度评估 · 7 个派生函数全代码

### 1.1 数据契约（每维只取 1-2 处）

| 维度 (FinOps 2026 命名) | 数据源 | 单一查询 |
|:---|:---|:---|
| **D1 delivery** | `Decisions/*.md` | grep `owner=actor, status=closed, closed_at>=since` · 看按时率 |
| **D2 cost_per_outcome** | `Trace/*.jsonl` | grep `actor=actor, ts>=since` · sum(value)/sum(cost) |
| **D3 quality** | `Decisions/*.md` | grep `owner=actor` · `1 - superseded/total` |
| **D4 judgment** | `Decisions/*.md` | grep `owner=actor, outcome!=null` · `OK%` |
| **D5 ecosystem** | `Skills/*.md` + `Trace/*.jsonl` | own Skills × distinct adopters |
| **D6 revenue_per_workflow** | `Decisions/*.md` + `value_anchors` | sum(roi_actual) OR anchor-weighted (0-90d) |
| **D7 learning** | `Trace/*.jsonl` | new skills used in window |

### 1.2 完整伪代码（Python）

```python
"""
.app/ome365_eval.py · v1.1 派生评估 · pure file derivation
GDPR Art. 22 compliant · all responses include human_review_required=True
FinOps 2026 aligned · D2 cost_per_outcome / D6 revenue_per_workflow
"""
from __future__ import annotations
from dataclasses import dataclass
from datetime import date, datetime, timedelta
from pathlib import Path
from math import log10, log2
from typing import Optional


@dataclass
class Score:
    raw: Optional[float]              # 原始指标值
    score: Optional[float]            # 0-5 normalized
    n: int                            # 样本数
    percentile: Optional[float]       # 团队内百分位 0-1
    reason: Optional[str] = None      # 拒绝原因 (insufficient_sample 等)


def _clip(x: float, lo: float, hi: float) -> float:
    return max(lo, min(hi, x))


# ── D1 交付力 ──────────────────────────────────────────────
def D1_delivery(actor: str, since: date, vault: Path, sample_min: int = 5) -> Score:
    """按时闭合率·planned_duration 来自决策 frontmatter 字段"""
    closed = grep_decisions(vault, owner=actor, status="closed", closed_at__gte=since)
    if len(closed) < sample_min:
        return Score(raw=None, score=None, n=len(closed), percentile=None,
                     reason="insufficient_sample")
    on_time = sum(1 for d in closed
                  if d.elapsed_days <= (d.planned_duration_days or float("inf")))
    raw = on_time / len(closed)               # 0-1
    score = _clip(raw * 5, 0, 5)              # 0-5
    return Score(raw=raw, score=score, n=len(closed), percentile=None)


# ── D2 cost_per_outcome ────────────────────────────────────
def D2_cost_per_outcome(actor: str, since: date, vault: Path, sample_min: int = 5) -> Score:
    """ROI multiple · sum(value) / sum(cost)"""
    traces = grep_trace(vault, actor=actor, ts__gte=since)
    if len(traces) < sample_min:
        return Score(raw=None, score=None, n=len(traces), percentile=None,
                     reason="insufficient_sample")
    cost = sum(t.cost_usd for t in traces)
    value = sum((t.output_value_usd or 0) for t in traces)
    if cost <= 0:
        return Score(raw=None, score=None, n=len(traces), percentile=None,
                     reason="zero_cost")
    raw = value / cost                        # ROI multiple
    # log scale: 1× → 0, 10× → 1, 100× → 2, 1000× → 3, 10k× → 4, 100k× → 5
    score = _clip(log10(raw + 1), 0, 5)
    return Score(raw=raw, score=score, n=len(traces), percentile=None)


# ── D3 质量守护 ──────────────────────────────────────────────
def D3_quality(actor: str, since: date, vault: Path, sample_min: int = 5) -> Score:
    """1 - 被 supersede 的反向比率"""
    decisions = grep_decisions(vault, owner=actor, closed_at__gte=since)
    if len(decisions) < sample_min:
        return Score(raw=None, score=None, n=len(decisions), percentile=None,
                     reason="insufficient_sample")
    superseded = sum(1 for d in decisions if d.superseded_by)
    raw = 1 - (superseded / len(decisions))
    score = _clip(raw * 5, 0, 5)
    return Score(raw=raw, score=score, n=len(decisions), percentile=None)


# ── D4 判断力 ──────────────────────────────────────────────
def D4_judgment(actor: str, since: date, vault: Path, sample_min: int = 5) -> Score:
    """outcome=OK 占比"""
    closed = grep_decisions(vault, owner=actor, status="closed", outcome__isnotnull=True)
    if len(closed) < sample_min:
        return Score(raw=None, score=None, n=len(closed), percentile=None,
                     reason="insufficient_sample")
    ok = sum(1 for d in closed
             if d.outcome and d.outcome.lower().startswith(("ok", "success", "成功")))
    raw = ok / len(closed)
    score = _clip(raw * 5, 0, 5)
    return Score(raw=raw, score=score, n=len(closed), percentile=None)


# ── D5 生态贡献（A4S 25% 权重最重）─────────────────────────
def D5_ecosystem(actor: str, since: date, vault: Path, sample_min: int = 1) -> Score:
    """own skills × distinct adopters · 防刷：排除 author=actor 的 trace"""
    own_skills = grep_skills(vault, author=actor, created__gte=since)
    skill_ids = {s.name for s in own_skills}
    if not skill_ids:
        return Score(raw=0, score=0, n=0, percentile=None)
    distinct_adopters: set[str] = set()
    for t in grep_trace(vault, ts__gte=since):
        if t.skill in skill_ids and t.actor != actor:    # 防 D5 自刷
            distinct_adopters.add(t.actor)
    raw = len(skill_ids) * len(distinct_adopters)        # composite
    # log2 scale: raw=0→0, raw=1→0, raw=4→1, raw=16→3, raw=64→5
    score = _clip(log2(raw + 1) - 1, 0, 5)
    return Score(raw=raw, score=score, n=len(skill_ids), percentile=None)


# ── D6 revenue_per_workflow ────────────────────────────────
def D6_revenue_per_workflow(actor: str, since: date, vault: Path,
                             value_anchor_weights: dict, sample_min: int = 3) -> Score:
    """两阶段：0-90 天用 value_anchors / 90 天后用 roi_actual"""
    decisions = grep_decisions(vault, owner=actor, status="closed", closed_at__gte=since)
    if len(decisions) < sample_min:
        return Score(raw=None, score=None, n=len(decisions), percentile=None,
                     reason="insufficient_sample")
    
    total_usd = 0.0
    anchor_score_sum = 0.0
    anchor_n = 0
    for d in decisions:
        days_since_close = (date.today() - d.closed_at.date()).days
        if days_since_close >= 90 and d.roi_actual is not None:
            total_usd += d.roi_actual
        elif d.value_anchors:                    # 0-90 天用锚点
            for anchor in d.value_anchors:
                anchor_score_sum += value_anchor_weights.get(anchor, 0)
                anchor_n += 1
    
    if total_usd > 0:
        # USD-based: $100→0, $10k→2, $1M→4, $100M→clip 5
        raw = total_usd
        score = _clip(log10(total_usd + 1) - 2, 0, 5)
    elif anchor_n > 0:
        # Anchor-based interim score (0-90 day window)
        raw = anchor_score_sum
        score = _clip(anchor_score_sum / anchor_n, 0, 5)
    else:
        return Score(raw=None, score=None, n=len(decisions), percentile=None,
                     reason="no_roi_data")
    return Score(raw=raw, score=score, n=len(decisions), percentile=None)


# ── D7 学习力 ──────────────────────────────────────────────
def D7_learning(actor: str, since: date, vault: Path) -> Score:
    """本期首次用的 skill 数"""
    used_in_window = {t.skill for t in grep_trace(vault, actor=actor, ts__gte=since)}
    used_before = {t.skill for t in grep_trace(vault, actor=actor, ts__lt=since)}
    new_skills = used_in_window - used_before
    raw = len(new_skills)
    score = _clip(raw, 0, 5)                  # 5+ = score 5
    return Score(raw=raw, score=score, n=len(used_in_window), percentile=None)


# ── 性能边界：team-distribution cache（Fix 3 · 5-08 review）────────
# N+1 problem: 原始 lazy 算 percentile 时·per-dim per-member 重跑 grep
# → 100 成员 × 7 dim = 700 vault scan · 1k decisions vault 单次 API 30-60s 必崩
# 修复: weekly batch + cache · O(N+V) 首次 / O(1) 命中 · cache 是派生 artifact (gitignored)

def _compute_team_distribution(tenant: str, since: date, vault: Path) -> dict:
    """ONE-PASS: build all 7 dims for all members in single vault scan · O(N+V)"""
    all_decisions = grep_decisions_all(vault, closed_at__gte=since)
    all_traces = grep_trace_all(vault, ts__gte=since)
    all_skills = grep_skills_all(vault)
    by_member = defaultdict(lambda: {"decisions": [], "traces": [], "skills": []})
    for d in all_decisions: by_member[d.owner]["decisions"].append(d)
    for t in all_traces:    by_member[t.actor]["traces"].append(t)
    for s in all_skills:    by_member[s.author]["skills"].append(s)
    return {mid: {f"D{i}_*": fn(data) for i, fn in enumerate([D1_delivery, ...], start=1)}
            for mid, data in by_member.items()}


def _load_or_compute_distribution(tenant, since, vault, max_age_hours=168) -> dict:
    """优先读 weekly cache · 过期则 recompute + 写盘"""
    period = since.strftime("%Y-W%V")
    cache_path = vault / "Eval" / "snapshots" / period / "team-distribution.json"
    if cache_path.exists():
        if (time.time() - cache_path.stat().st_mtime) / 3600 < max_age_hours:
            return json.loads(cache_path.read_text())
    distribution = _compute_team_distribution(tenant, since, vault)
    cache_path.parent.mkdir(parents=True, exist_ok=True)
    cache_path.write_text(json.dumps(distribution, default=str))
    return distribution


# ── eval_member 入口 ───────────────────────────────────────
def eval_member(tenant: str, member_id: str, *,
                window_days: int = 30,
                cfg_path: Optional[Path] = None,
                use_cache: bool = True) -> dict:
    vault = vault_root(tenant)
    since = date.today() - timedelta(days=window_days)
    cfg = load_eval_config(cfg_path or vault / ".ome365" / "eval-config.yml")
    
    # Region-aware enforcement (Fix 5 · 2026-05-08 · GDPR + PIPL 双轨)
    region = cfg.get("region", "global")  # cn | eu | us | global
    if region == "eu" and not cfg.get("eval_enabled_eu", False):
        raise EvalDisabledForRegion("EU · GDPR Art. 22 · default-deny")
    if region == "cn":
        # PIPL §13 自动告知 + §24 自动决策需个人书面同意
        ack = cfg.get("pipl_notify_acknowledged_by", {})
        if member_id not in ack:
            raise PIPLNotifyRequired(f"member {member_id} · PIPL §13 告知未签 · "
                                     f"see docs/legal/PIPL-NOTIFY-TEMPLATE.zh.md")
        if member_id in cfg.get("opted_out_members", []):
            raise EvalOptedOut(f"member {member_id} · PIPL §24 拒绝权已行使")
    # us / global region: no extra check · default open · still warn human_review_required
    
    weights = cfg["weights"]
    anchor_weights = cfg["value_anchor_weights"]
    sample_min = cfg.get("sample_size_min", 5)
    
    dims = {
        "D1_delivery":             D1_delivery(member_id, since, vault, sample_min),
        "D2_cost_per_outcome":     D2_cost_per_outcome(member_id, since, vault, sample_min),
        "D3_quality":              D3_quality(member_id, since, vault, sample_min),
        "D4_judgment":             D4_judgment(member_id, since, vault, sample_min),
        "D5_ecosystem":            D5_ecosystem(member_id, since, vault, sample_min=1),
        "D6_revenue_per_workflow": D6_revenue_per_workflow(member_id, since, vault,
                                                           anchor_weights, sample_min=3),
        "D7_learning":             D7_learning(member_id, since, vault),
    }
    
    # Compute percentile within team (lazy · scan all team members)
    team_members = list_team_members(tenant, member_id_self=member_id)
    for k, my_score in dims.items():
        if my_score.score is None:
            continue
        team_scores = []
        for tm in team_members:
            ts = globals()[f"{k.split('_')[0]}_{k.split('_', 1)[1]}"](
                tm, since, vault, sample_min)
            if ts.score is not None:
                team_scores.append(ts.score)
        if team_scores:
            below = sum(1 for s in team_scores if s < my_score.score)
            my_score.percentile = below / len(team_scores)
    
    # Weighted total (only valid dims)
    valid = {k: v for k, v in dims.items() if v.score is not None}
    if valid:
        weight_sum = sum(weights.get(k.split("_", 1)[1], 0) for k in valid.keys())
        total = sum(v.score * weights.get(k.split("_", 1)[1], 0)
                    for k, v in valid.items()) / max(weight_sum, 1e-9)
    else:
        total = None
    
    return {
        "tenant": tenant,
        "member_id": member_id,
        "window_days": window_days,
        "dimensions": {k: v.__dict__ for k, v in dims.items()},
        "total_score": total,
        "weights_used": weights,
        "human_review_required": True,           # GDPR Art. 22 强制
        "warning": "Eval is for resource allocation hint only. NEVER as sole basis for HR decisions.",
        "computed_at": datetime.utcnow().isoformat() + "Z",
    }
```

```python
def backfill_outcomes(vault: Path):
    """Nightly · 找已闭合 ≥90 天且 roi_actual=null 的决策"""
    cutoff = date.today() - timedelta(days=90)
    pending = grep_decisions(vault, status="closed",
                             closed_at__lt=cutoff, roi_actual=None)
    
    for d in pending:
        # 1. 找相关 Trace
        traces = grep_trace(vault, decision_id=d.id)
        # 2. 找后续被 supersede（说明可能错）
        successors = grep_decisions(vault, supersedes=d.id)
        # 3. LLM 起草 ⑧ Feedback
        prompt = DECISION_FEEDBACK_PROMPT.format(
            decision=d, traces=traces, successors=successors,
            calibration_examples=load_recent_calibrations(vault, limit=5))
        draft = llm_chat(prompt, model="claude-haiku-4.5")
        # 4. 不直接写盘 · 写到 .proposals/
        proposal_path = vault / "Decisions" / ".proposals" / f"{d.id}-feedback.md"
        proposal_path.write_text(draft)
        # 5. 通知 owner
        notify(d.owner, f"Decision {d.id} feedback proposal ready · review at {proposal_path}")
```

```python
def distill_to_l2(vault: Path):
    """每周日 02:00 · 把最近 7 天闭合决策按 category 聚合"""
    closed_recent = grep_decisions(vault, status="closed",
                                    closed_at__gte=date.today() - timedelta(days=7))
    by_category = defaultdict(list)
    for d in closed_recent:
        cat = d.category or extract_category_from_title(d.title)
        by_category[cat].append(d)
    
    for cat, group in by_category.items():
        if len(group) < 3:
            continue                      # 不到 3 不聚合
        prompt = DISTILL_PATTERN_PROMPT.format(decisions=group,
                                                calibration_examples=load_recent_calibrations(vault))
        pattern_md = llm_chat(prompt, model="claude-sonnet-4.5")
        target = vault / "Knowledge" / "L2-distilled" / "decision-patterns" / f"{cat}.md"
        target.parent.mkdir(parents=True, exist_ok=True)
        # append 而不是覆盖
        with target.open("a") as f:
            f.write(f"\n\n## Pattern · {date.today().isoformat()}\n\n{pattern_md}\n")
```

## 三 · 知识共享 · 4 级 privacy + 跨租户

### 3.1 privacy 4 级语义

| privacy | 谁能看 | 跨 tenant | 例子 |
|:---:|:---|:---:|:---|
| `public` | 全员 + 外部全网 | ✅ | 公开 best practice · 公共 Skill |
| `pinned` | 全员 + 白名单外部 tenant DID | ✅ T2 | 战略合作伙伴共享 |
| `internal` | 同 tenant 全员 | ❌ | 大部分 L2 patterns |
| `sensitive` | 同 tenant + 显式 ACL | ❌ | 个人特征评估 · 未公开决策 |

### 3.2 can_see 函数（核心权限判断）

```python
def can_see(file_privacy: str, *,
            file_path: Path,
            caller_tenant: str,
            caller_did: str,
            current_tenant: str,
            pinlist: list[str],
            file_acl: list[str] = None) -> tuple[bool, str]:
    """
    Returns (allowed, reason).
    跨 tenant 时 caller_tenant != current_tenant.
    """
    if file_privacy == "public":
        return True, "public · open to all"
    
    if file_privacy == "pinned":
        if caller_did in pinlist:
            return True, f"pinned · {caller_did} in pinlist"
        return False, f"pinned · {caller_did} not in pinlist"
    
    if file_privacy == "internal":
        if caller_tenant == current_tenant:
            return True, "internal · same tenant"
        return False, f"internal · cross-tenant denied ({caller_tenant} != {current_tenant})"
    
    if file_privacy == "sensitive":
        if caller_tenant != current_tenant:
            return False, "sensitive · cross-tenant always denied"
        if file_acl and caller_did in file_acl:
            return True, "sensitive · in ACL"
        return False, "sensitive · ACL miss"
    
    return False, f"unknown privacy: {file_privacy}"
```

```python
def should_keep(file_path: Path, vault: Path) -> tuple[bool, dict]:
    """评分模型：5+ 通过 · 5% 保留率"""
    score = 0
    reasons = []
    
    # 最近 7 天访问
    if last_modified_or_accessed(file_path) >= date.today() - timedelta(days=7):
        score += 5
        reasons.append("recently_accessed")
    
    # 至少 2 处引用
    refs = count_references(file_path, vault)
    if refs >= 2:
        score += 3
        reasons.append(f"references_{refs}")
    
    # 被决策 link
    if linked_from_decision(file_path, vault):
        score += 2
        reasons.append("linked_from_decision")
    
    # 高 adoption（Skills 适用）
    if file_path.parent.name == "Skills":
        adopters = count_skill_adopters(file_path, vault, days=30)
        if adopters >= 3:
            score += 4
            reasons.append(f"adopters_{adopters}")
    
    return score >= 5, {"score": score, "reasons": reasons}


def archive_weekly(vault: Path):
    """每周日 02:00 · 扫 Notes/ Knowledge/L1/ Decisions/ 三处"""
    archived_count = 0
    kept_count = 0
    log_path = vault / ".ome365" / "archive-log" / f"{date.today().isoformat()}.md"
    log_path.parent.mkdir(parents=True, exist_ok=True)
    
    with log_path.open("w") as log:
        log.write(f"# Archive log · {date.today()}\n\n")
        for folder in ["Notes", "Knowledge/L1-raw", "Decisions"]:
            folder_path = vault / folder
            if not folder_path.exists():
                continue
            for f in folder_path.rglob("*.md"):
                if ".archive" in f.parts:
                    continue           # already archived
                keep, info = should_keep(f, vault)
                if keep:
                    kept_count += 1
                else:
                    period = f"{date.today().year}-W{date.today().isocalendar()[1]:02d}"
                    target = folder_path / ".archive" / period / f.name
                    target.parent.mkdir(parents=True, exist_ok=True)
                    f.rename(target)
                    archived_count += 1
                    log.write(f"- archive {f.name} · score={info['score']}\n")
        log.write(f"\n## Summary\nArchived: {archived_count}\nKept: {kept_count}\n")
        log.write(f"Archive ratio: {archived_count / max(archived_count + kept_count, 1):.1%}\n")
```

### 4.2 Strategy 1: Proportional（v1.1 默认）

```python
def attribute_value_proportional(vault: Path, decision_id: str):
    """Decision 闭合 + roi_actual 回填后调用 · value 按 cost 比例分摊到 trace"""
    d = read_decision(vault, decision_id)
    if d.roi_actual is None:
        return                                       # 还没回填
    
    traces = grep_trace_files(vault, decision_id=decision_id)
    total_cost = sum(t.cost_usd for t in traces if t.value_attribution is None)
    if total_cost == 0:
        return                                       # 无可归因 trace
    
    for t in traces:
        if t.value_attribution is not None:
            continue                                 # 已归因·跳
        share = t.cost_usd / total_cost
        t.output_value_usd = d.roi_actual * share
        t.value_attribution = "proportional"
        update_jsonl_line(t.file, t.line_no, t)      # in-place update
```

```python
def attribute_value_llm(vault: Path, decision_id: str, model: str = "claude-haiku-4.5"):
    """让 LLM 给每条 trace 0-1 贡献度 · 适用高 value decision · 贵但准"""
    d = read_decision(vault, decision_id)
    traces = grep_trace_files(vault, decision_id=decision_id)
    
    prompt = LLM_ATTRIBUTION_PROMPT.format(
        decision_summary=d.outcome_summary,
        traces=[{"ts": t.ts, "skill": t.skill, "summary": t.summary} for t in traces])
    contributions = llm_chat(prompt, model=model)    # returns dict {trace_id: weight 0-1}
    
    total_weight = sum(contributions.values())
    if total_weight == 0:
        return
    for t in traces:
        weight = contributions.get(t.id, 0)
        t.output_value_usd = d.roi_actual * (weight / total_weight)
        t.value_attribution = "llm_judged"
        update_jsonl_line(t.file, t.line_no, t)
```

### 4.4 finops_summary · 三视图

```python
def finops_summary(tenant: str, *,
                   scope: str,
                   since: Optional[date] = None) -> dict:
    """
    scope: cost_per_resolved_decision | human_equivalent_hourly | revenue_per_workflow
    """
    vault = vault_root(tenant)
    traces = grep_trace_all(vault, ts__gte=since)
    
    if scope == "cost_per_resolved_decision":
        decisions_closed = grep_decisions(vault, status="closed", closed_at__gte=since)
        cost = sum(t.cost_usd for t in traces)
        n_decisions = len(decisions_closed)
        return {"scope": scope, "value": cost / max(n_decisions, 1),
                "unit": "USD per closed decision",
                "human_review_required": True}
    
    if scope == "human_equivalent_hourly":
        cost = sum(t.cost_usd for t in traces)
        # hours_saved 来自 decision.metadata.hours_saved（owner 闭合时填）
        decisions_closed = grep_decisions(vault, status="closed", closed_at__gte=since)
        hours_saved = sum((d.hours_saved or 0) for d in decisions_closed)
        return {"scope": scope, "value": cost / max(hours_saved, 1e-9),
                "unit": "USD per human hour saved",
                "human_review_required": True}
    
    if scope == "revenue_per_workflow":
        # workflow = 1 个 decision execution
        decisions_closed = grep_decisions(vault, status="closed", closed_at__gte=since)
        total_value = sum((d.roi_actual or 0) for d in decisions_closed
                          if d.roi_actual is not None and d.roi_actual > 0)
        n_workflows = len(decisions_closed)
        return {"scope": scope, "value": total_value / max(n_workflows, 1),
                "unit": "USD per workflow execution",
                "human_review_required": True}
    
    raise ValueError(f"unknown scope: {scope}")
```
